# 서비스(Service) 계층

[[04_레이어드_아키텍처]]에서 Router와 Repository 사이에 있던 계층을 구체화한 것이다.

## 개념

요청에 해당하는 비즈니스 로직 처리를 담당하는 계층이다. DB 정보가 필요하면 직접 쿼리를 짜지 않고 [[12_리포지토리_패턴]]에서 만든 리포지토리를 호출한다.

**서비스 함수 하나가 곧 트랜잭션 하나의 단위다.** 서비스 함수 안에서 리포지토리를 여러 번 호출할 수 있고, 그 호출들이 전부 성공해야 commit되고, 하나라도 실패하면 전부 rollback된다 — [[11_SQLAlchemy]]에서 만든 `session_scope()`가 정확히 이 "시작~성공 시 commit~실패 시 rollback" 경계를 담당한다.

## 실습: `app/services/document_service.py`

```python
"""
서비스
- 사용자 요청에 대한 로직 처리 하는 곳
- DB 정보가 필요하면 repository 호출
- 트랜잭션을 여닫는 자리
"""
from __future__ import annotations

from datetime import date

from app.core.exceptions import NotFound, ValidationFailed
from app.db.session import session_scope
from app.models.document import Document, DocumentVersion
from app.repositories import document_repo

# 여러 객체(DocumentVersion, Document)에 나뉘어 있는 데이터를 응답용 딕셔너리 하나로 묶는다
def _to_out(version: DocumentVersion, document: Document) -> dict:
    return {
        "doc_id": document.id,
        "title": document.title,
        "dept": document.dept.name,
        "version": version.version,
        "security_level": document.security_level,
        "file_format": version.file_format,
        "status": version.status,
        "effective_from": version.effective_from,
        "expires_at": version.expires_at,
        "index_status": version.index_status,
        "index_progress": version.index_progress,
    }

def list_documents(
    *,
    dept_id: str | None = None,
    security_level: str | None = None,
    status: str | None = None,
    q: str | None = None,
    limit: int = 50,
) -> list[dict]:
    with session_scope() as s:
        rows = document_repo.list_documents(
            s, dept_id=dept_id, security_level=security_level,
            status=status, q=q, limit=limit,
        )
        return [_to_out(version, document) for version, document in rows]

def get_document(*, doc_id: str) -> dict:
    with session_scope() as s:
        document = document_repo.get_document(s, doc_id)
        if document is None:
            raise NotFound(f"문서를 찾을 수 없습니다: {doc_id}")
        current = document.current
        if current is None:
            raise NotFound(f"현행 버전이 없습니다: {doc_id}")
        return _to_out(current, document)

def create_document(
    *,
    doc_id: str,
    title: str,
    dept_id: str,
    security_level: str,
    version: str,
    effective_from: date,
    file_path: str,
    file_format: str,
    owner_id: int | None = None,
) -> dict:
    with session_scope() as s:
        document = document_repo.get_document(s, doc_id)
        created = document is None

        if document is None:
            # 처음 등록하는 문서면 Document부터 새로 만든다
            document = Document(
                id=doc_id, title=title, dept_id=dept_id,
                security_level=security_level, owner_id=owner_id,
            )
            s.add(document)
            s.flush()
        elif any(v.version == version for v in document.versions):
            # 이미 있는 문서인데 같은 버전을 또 등록하려 하면 막는다
            raise ValidationFailed(
                f"{doc_id} 의 {version} 은(는) 이미 등록되어 있습니다. "
                "판 번호를 올려 다시 올려 주세요."
            )

        document_repo.add_version(
            s, document, version=version, status="현행",
            effective_from=effective_from, expires_at=None,
            file_path=file_path, file_format=file_format,
            index_status="대기",
        )
        return {
            "doc_id": document.id, "title": document.title, "version": version,
            "file_format": file_format, "file_path": file_path, "created": created,
        }
```

- `with session_scope() as s:` 블록 하나가 트랜잭션 하나다. `create_document()` 안에서 `document_repo.get_document()`(조회), `s.add(document)`(문서 생성), `document_repo.add_version()`(버전 추가)까지 리포지토리를 여러 번 부르지만, 이게 전부 같은 세션(`s`) 안에서 일어나고 함수를 빠져나갈 때 `session_scope()`가 한 번에 commit(또는 실패 시 rollback)한다.
- `raise NotFound(...)`, `raise ValidationFailed(...)`처럼 [[05_예외_계층_설계]]에서 만든 커스텀 예외를 여기서 던진다. `session_scope()`의 `except Exception: session.rollback(); raise`가 이 예외를 받아서 롤백부터 하고 그대로 다시 던지므로, 예외가 발생한 시점까지의 변경사항(예: 새로 만든 `Document`)은 저장되지 않는다.
- `_to_out()`처럼 "DB 모델 객체 → 응답용 딕셔너리" 변환을 서비스 계층에 두는 이유는, 라우터가 SQLAlchemy 모델 객체를 몰라도 되게 하기 위해서다 ([[10_FastAPI]]의 요청/응답 스키마 분리와 같은 맥락).
- `document is None`으로 신규/기존을 나누고, 기존 문서인데 같은 버전이면 막는 로직처럼 "이게 되는 조합인지 아닌지" 판단하는 게 전형적인 서비스 계층의 일이다. 리포지토리는 "조회해라/저장해라"만 할 뿐 이런 판단을 하지 않는다.

## 계층별 책임 정리

| 계층 | 아는 것 | 모르는 것 |
| --- | --- | --- |
| Router | HTTP 요청/응답, 쿼리 파라미터 | SQL, 트랜잭션 |
| Service | 비즈니스 규칙, 트랜잭션 경계 | SQL 문법 자체 |
| Repository | SQL/쿼리 작성법 | 이 조회 결과로 뭘 판단할지 |
| Model | 테이블 구조, 컬럼 간 관계 | 이 데이터가 어떤 요청에서 쓰이는지 |
